Ejemplo: Test de Composable

// tests/unit/composables/useBooks.test.js
import { describe, it, expect, beforeEach, vi } from 'vitest'
import { useBooks } from '@/composables/useBooks'
import * as api from '@/services/api'

// Mock API
vi.mock('@/services/api')

describe('useBooks', () => {
  beforeEach(() => {
    vi.clearAllMocks()
  })

  it('should fetch books successfully', async () => {
    // Arrange
    const mockBooks = [
      { id: 1, title: 'Test Book', author: 'Test Author' }
    ]
    api.getBooks.mockResolvedValue({ data: mockBooks })

    // Act
    const { books, fetchBooks, isLoading } = useBooks()
    await fetchBooks()

    // Assert
    expect(isLoading.value).toBe(false)
    expect(books.value).toEqual(mockBooks)
    expect(api.getBooks).toHaveBeenCalledOnce()
  })

  it('should handle fetch errors', async () => {
    // Arrange
    const errorMessage = 'Network error'
    api.getBooks.mockRejectedValue(new Error(errorMessage))

    // Act
    const { error, fetchBooks } = useBooks()
    await fetchBooks()

    // Assert
    expect(error.value).toContain(errorMessage)
  })
})

Ejemplo: Test de Componente

// tests/unit/components/BookCard.test.js
import { describe, it, expect } from 'vitest'
import { mount } from '@vue/test-utils'
import BookCard from '@/components/BookCard.vue'

describe('BookCard', () => {
  const mockBook = {
    id: 1,
    title: 'Harry Potter',
    author: 'J.K. Rowling',
    status: 'read',
    user_rating: 5
  }

  it('renders book information correctly', () => {
    const wrapper = mount(BookCard, {
      props: { book: mockBook }
    })

    expect(wrapper.text()).toContain('Harry Potter')
    expect(wrapper.text()).toContain('J.K. Rowling')
    expect(wrapper.find('.rating-stars').exists()).toBe(true)
  })

  it('emits edit event when edit button is clicked', async () => {
    const wrapper = mount(BookCard, {
      props: { book: mockBook }
    })

    await wrapper.find('.edit-button').trigger('click')
    
    expect(wrapper.emitted().edit).toBeTruthy()
    expect(wrapper.emitted().edit[0]).toEqual([mockBook])
  })
})

Ejemplo: Test de Use Case

<?php
// tests/Unit/UseCases/AddBookUseCaseTest.php

namespace Tests\Unit\UseCases;

use PHPUnit\Framework\TestCase;
use Mockery as m;
use App\Domain\UseCases\AddBookUseCase;
use App\Domain\Repository\BookRepositoryInterface;

class AddBookUseCaseTest extends TestCase
{
    private $bookRepository;
    private $addBookUseCase;

    protected function setUp(): void
    {
        $this->bookRepository = m::mock(BookRepositoryInterface::class);
        $this->addBookUseCase = new AddBookUseCase($this->bookRepository);
    }

    protected function tearDown(): void
    {
        m::close();
    }

    public function testExecuteSuccessfully()
    {
        // Arrange
        $bookData = [
            'title' => 'Test Book',
            'author' => 'Test Author',
            'isbn' => '9781234567890'
        ];
        $userId = 1;

        $this->bookRepository
            ->shouldReceive('addUserBook')
            ->once()
            ->with($userId, $bookData)
            ->andReturn(['id' => 123] + $bookData);

        // Act
        $result = $this->addBookUseCase->execute($userId, $bookData);

        // Assert
        $this->assertEquals(123, $result['id']);
        $this->assertEquals('Test Book', $result['title']);
    }

    public function testExecuteWithInvalidData()
    {
        // Arrange
        $invalidData = ['title' => '']; // Missing required fields
        $userId = 1;

        // Act & Assert
        $this->expectException(\InvalidArgumentException::class);
        $this->addBookUseCase->execute($userId, $invalidData);
    }
}

Ejemplo: Test de API

<?php
// tests/Integration/API/BooksAPITest.php

namespace Tests\Integration\API;

use Tests\TestCase;

class BooksAPITest extends TestCase
{
    protected function setUp(): void
    {
        parent::setUp();
        $this->authenticateUser();
    }

    public function testGetBooksReturnsUserBooks()
    {
        // Arrange
        $this->createTestBooks();

        // Act
        $response = $this->get('/api/books');

        // Assert
        $response->assertStatus(200);
        $response->assertJsonStructure([
            'success',
            'data' => [
                'books' => [
                    '*' => ['id', 'title', 'author', 'status']
                ],
                'pagination'
            ]
        ]);
    }

    public function testAddBookSuccessfully()
    {
        // Arrange
        $bookData = [
            'title' => 'New Test Book',
            'author' => 'Test Author',
            'isbn' => '9781234567890'
        ];

        // Act
        $response = $this->post('/api/books', $bookData);

        // Assert
        $response->assertStatus(201);
        $response->assertJson([
            'success' => true,
            'data' => [
                'title' => 'New Test Book'
            ]
        ]);

        $this->assertDatabaseHas('user_books', [
            'title' => 'New Test Book',
            'user_id' => $this->user->id
        ]);
    }
}

Ejecutar Tests

Frontend
# Todos los tests
npm run test

# Tests en modo watch
npm run test:watch

# Tests con coverage
npm run test:coverage

# E2E tests
npm run test:e2e

# Tests específicos
npm run test -- BookCard.test.js
Backend
# Todos los tests
./vendor/bin/phpunit

# Tests con coverage
./vendor/bin/phpunit --coverage-html coverage

# Tests específicos
./vendor/bin/phpunit tests/Unit/UseCases

# Tests de integración
./vendor/bin/phpunit tests/Integration
Docker Testing
# Tests en contenedores Docker
docker-compose exec frontend npm run test
docker-compose exec backend ./vendor/bin/phpunit

# Setup de testing database
docker-compose exec backend php artisan migrate:fresh --env=testing

Mejores Prácticas

✅ Hacer
  • Tests pequeños y enfocados
  • Nombres descriptivos
  • Arrange-Act-Assert pattern
  • Mock dependencias externas
  • Tests independientes
  • Cleanup después de tests
❌ Evitar
  • Tests que dependen del orden
  • Hard-coded values
  • Tests que tocan servicios reales
  • Tests demasiado complejos
  • Duplicación de lógica de tests
  • Aserciones múltiples por concepto
Estructura de Test Recomendada
describe('Component/Function Name', () => {
  // Setup
  beforeEach(() => {
    // Common setup
  })

  describe('when condition', () => {
    it('should do expected behavior', () => {
      // Arrange - Setup test data
      const input = 'test data'
      
      // Act - Execute the function/action
      const result = functionUnderTest(input)
      
      // Assert - Verify results
      expect(result).toBe('expected output')
    })
  })
})

GitHub Actions Ejemplo

# .github/workflows/test.yml
name: Tests

on: [push, pull_request]

jobs:
  frontend-tests:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      - uses: actions/setup-node@v3
        with:
          node-version: '18'
      - run: cd frontend && npm ci
      - run: cd frontend && npm run test:coverage
      
  backend-tests:
    runs-on: ubuntu-latest
    services:
      mysql:
        image: mysql:8.0
        env:
          MYSQL_ROOT_PASSWORD: password
    steps:
      - uses: actions/checkout@v3
      - uses: shivammathur/setup-php@v2
        with:
          php-version: '8.1'
      - run: cd backend && composer install
      - run: cd backend && ./vendor/bin/phpunit
